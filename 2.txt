import re

def validate_credit_card(card):
    # 1. Check format: starts with 4,5,6; 16 digits; optional hyphens every 4 digits
    # Pattern: ^[456] followed by 3 digits, then 3 groups of (-digit{4}) or (digit{4})
    format_re = r'^[456]\d{3}(-?\d{4}){3}$'
    
    # 2. Check for 4+ consecutive repeated digits (ignoring hyphens)
    # Remove hyphens first to check actual digit sequence
    clean_card = card.replace('-', '')
    repetition_re = r'(\d)\1{3,}'
    
    if re.match(format_re, card) and not re.search(repetition_re, clean_card):
        return "Valid"
    else:
        return "Invalid"

# Process input
n = int(input())
for _ in range(n):
    print(validate_credit_card(input().strip()))
