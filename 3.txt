# Enter your code here. Read input from STDIN. Print output to STDOUT
import re

def validate_uid(uid):
    # Rule 5: Exactly 10 characters
    if len(uid) != 10:
        return False
    
    # Rule 3: Only alphanumeric
    if not uid.isalnum():
        return False
    
    # Rule 4: No character repeats
    if len(set(uid)) != 10:
        return False
    
    # Rule 1: At least 2 uppercase
    if len(re.findall(r'[A-Z]', uid)) < 2:
        return False
    
    # Rule 2: At least 3 digits
    if len(re.findall(r'\d', uid)) < 3:
        return False
    
    return True

# Input handling
import sys
input_data = sys.stdin.read().split()
t = int(input_data[0])
for i in range(1, t + 1):
    if validate_uid(input_data[i]):
        print("Valid")
    else:
        print("Invalid")
